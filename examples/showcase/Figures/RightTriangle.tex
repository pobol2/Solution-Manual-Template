    \coordinate (C) at (0, 0);
    \coordinate (B) at (4, 0);
    \coordinate (A) at (0, 3);

    \fill[region] (A) -- (B) -- (C) -- cycle;
    \draw[highlight] (A) -- node[above right] {$c$} (B);
    \draw[thick] (B) -- (C) -- (A);
    \draw pic[right angle mark] {right angle = A--C--B};
    \draw pic[angle arc, "$\theta$"] {angle = A--B--C};

    \draw[dimension] ([xshift=-4mm]C) -- node[left] {$a = 3$} ([xshift=-4mm]A);
    \draw[dimension] ([yshift=-4mm]C) -- node[below] {$b = 4$} ([yshift=-4mm]B);

    \foreach \p/\pos in {A/above, B/right, C/below left}
        \node[point, label=\pos:$\p$] at (\p) {};
